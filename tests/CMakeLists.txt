cmake_minimum_required(VERSION 3.16)
project(mixos_host_tests C)
set(CMAKE_C_STANDARD 11)
enable_testing()
set(ESP ../firmware/esp32s3/main)
add_executable(test_core test_core.c ${ESP}/mix_protocol.c ${ESP}/mix_terminal.c)
target_include_directories(test_core PRIVATE ${ESP})
# Clang on Windows can use the GNU-style driver while still targeting the
# Windows linker.  Detect the platform rather than the compiler front-end so
# we do not pass the Unix-only libm name to lld-link.
if(WIN32)
  target_compile_options(test_core PRIVATE -Wall -Wextra -Wpedantic)
else()
  target_compile_options(test_core PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(test_core m)
endif()
add_test(NAME core COMMAND test_core)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/test_input.c")
  add_executable(test_input test_input.c ${ESP}/mix_input.c)
  target_include_directories(test_input PRIVATE ${ESP})
  add_test(NAME input COMMAND test_input)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/test_terminal.c")
  add_executable(test_terminal test_terminal.c ${ESP}/mix_terminal.c)
  target_include_directories(test_terminal PRIVATE ${ESP})
  target_compile_options(test_terminal PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME terminal COMMAND test_terminal)
endif()
